📊 NEXT STEP: DAILY AGGREGATION
========================================

SELECT
    f.dt_posted::date AS date,
    a.general_issue AS canonical_issue,
    SUM(
        COALESCE(f.income, f.expenses)
        / COUNT(*) OVER (PARTITION BY a.filing_uuid)
    ) AS daily_spend
FROM lobbying_activity_lda a
JOIN lobbying_filing_lda f
    ON a.filing_uuid = f.filing_uuid
WHERE COALESCE(f.income, f.expenses) IS NOT NULL
GROUP BY 1, 2
ORDER BY 1, 2;
